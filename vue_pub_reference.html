// Vue에서 필요한 기능들을 import
// ref — 반응형 변수 선언 (값 바뀌면 화면 자동 업데이트)
// onMounted — 컴포넌트가 DOM에 그려진 후 실행
// onUnmounted — 컴포넌트가 사라지기 전 실행 (정리 작업)
import { ref, onMounted, onUnmounted } from 'vue'

// 메뉴 데이터 — 실제 프로젝트에서는 API나 별도 파일에서 가져올 수도 있음
// children이 있으면 드롭다운 메뉴로 처리
const menus = [
  { id: 1, label: '서비스 소개', path: '/service', children: [] },
  { id: 2, label: '솔루션',    path: '/solution',
    children: [
      { id: 21, label: '솔루션 A', path: '/solution/a' },
      { id: 22, label: '솔루션 B', path: '/solution/b' },
    ]
  },
  { id: 3, label: '공지사항', path: '/notice', children: [] },
  { id: 4, label: '문의하기', path: '/contact', children: [] },
]

// ref(초기값) — 반응형 변수 선언
// isScrolled: 스크롤 80px 넘으면 true로 바꿔서 헤더 스타일 변경
// isMobileOpen: 모바일 메뉴 열림/닫힘 상태
// currentHover: 현재 hover된 메뉴 id (드롭다운 제어)
const isScrolled    = ref(false)
const isMobileOpen  = ref(false)
const currentHover  = ref(null)

// /public 폴더의 이미지는 /로 시작하는 절대경로 사용
// /src/assets 이미지는 import 사용 (빌드 시 최적화)
const logoImg = '/images/logo.svg'

// 일반 함수 — window.scrollY로 현재 스크롤 위치 확인
// isScrolled.value = ... 으로 ref 값 수정 (.value 필수)
function handleScroll() {
  isScrolled.value = window.scrollY > 80
}

// onMounted — 컴포넌트가 DOM에 붙은 직후 실행
// 스크롤 이벤트 리스너 등록
onMounted(() => {
  window.addEventListener('scroll', handleScroll)
})

// onUnmounted — 컴포넌트가 사라질 때 반드시 이벤트 제거
// 안 하면 메모리 누수 발생! 컴포넌트가 없어도 이벤트가 계속 실행됨
onUnmounted(() => {
  window.removeEventListener('scroll', handleScroll)
})

💡
slot을 사용하면 모달 안에 들어갈 내용을 페이지마다 자유롭게 바꿀 수 있어요. 헤더/본문/푸터 영역을 named slot으로 나눠두면 유연하게 재사용할 수 있어요.
📄 BaseModal.vue — template
<!-- Transition — Vue 내장 컴포넌트
     v-if/v-show 요소가 나타나고 사라질 때 자동으로 CSS 클래스를 붙여줌
     name="modal" → .modal-enter-from, .modal-leave-to 등이 됨 -->
<Transition name="modal">

  <!-- v-if="modelValue" — 부모에서 v-model로 연결된 값
       true면 모달 표시, false면 DOM에서 제거
       modelValue는 v-model의 기본 prop 이름 -->
  <div
    v-if="modelValue"
    class="modal-overlay"

    <!-- @click.self — 자기 자신(오버레이)을 직접 클릭했을 때만 실행
         .self 없으면 모달 내부를 클릭해도 닫혀버림!
         $emit('update:modelValue', false) — 부모의 변수를 false로 바꿔서 닫음 -->
    @click.self="$emit('update:modelValue', false)"
  >

    <!-- :style — 인라인 스타일 동적 바인딩
         props로 받은 width 값을 스타일에 적용
         컴포넌트 사용할 때마다 다른 너비 지정 가능 -->
    <div class="modal" :style="{ width: width }">

      <!-- @click — 닫기 버튼 클릭 시 emit
           v-model과 연결된 부모 변수를 false로 변경해서 모달 닫음 -->
      <button
        class="modal__close"
        @click="$emit('update:modelValue', false)"
        aria-label="닫기"
      >✕</button>

      <!-- $slots.header — 부모가 #header slot에 내용을 넣었는지 확인
           내용이 있을 때만 헤더 영역 렌더링 -->
      <div v-if="$slots.header" class="modal__header">
        <!-- slot name="header" — 이름 있는 슬롯
             부모에서 <template #header>로 채워넣음 -->
        <slot name="header" />
      </div>

      <div class="modal__body">
        <!-- 이름 없는 기본 slot — 부모가 태그 사이에 넣은 내용이 여기에 들어감 -->
        <slot />
      </div>

      <div v-if="$slots.footer" class="modal__footer">
        <slot name="footer" />
      </div>

    </div>
  </div>
</Transition>
⚙️ BaseModal.vue — script setup
import { defineProps, defineEmits } from 'vue'

// defineProps — 부모에서 받을 데이터 정의
// modelValue: v-model과 연결되는 기본 prop 이름, Boolean 타입
// width: 모달 너비 커스터마이징용, 기본값 '480px'
defineProps({
  modelValue: Boolean,
  width: { type: String, default: '480px' }
})

// defineEmits — 부모로 전달할 이벤트 선언
// 'update:modelValue' — v-model을 업데이트하는 Vue 규칙
// 부모에서 v-model="isOpen" 하면 이 emit으로 isOpen 값이 바뀜
defineEmits(['update:modelValue'])

📄 TabSection.vue — template
<section class="tab-section">

  <ul class="tab-list" role="tablist">

    <!-- v-for로 탭 버튼 반복 생성
         :class 객체 방식 — is-active는 currentTab이 일치할 때만 붙음 -->
    <li
      v-for="tab in tabs"
      :key="tab.id"
      class="tab-item"
      :class="{ 'is-active': currentTab === tab.id }"

      <!-- @click — 클릭하면 currentTab 변경
           currentTab이 바뀌면 아래 computed가 자동으로 재계산됨 -->
      @click="currentTab = tab.id"
    >
      {{ tab.label }}
    </li>

  </ul>

  <div class="tab-content">

    <!-- Transition name="fade" mode="out-in"
         mode="out-in" — 이전 컴포넌트가 완전히 사라진 후 새 것이 나타남
         사라지고 나타나는 게 겹치지 않아서 자연스러움 -->
    <Transition name="fade" mode="out-in">

      <!-- component :is — 동적 컴포넌트
           currentComponent computed 값에 따라 렌더링할 컴포넌트가 바뀜
           :key — currentTab이 바뀌면 컴포넌트를 강제 재생성해서 Transition 발동 -->
      <component
        :is="currentComponent"
        :key="currentTab"
      />

    </Transition>
  </div>

</section>
⚙️ TabSection.vue — script setup
import { ref, computed } from 'vue'
import TabService   from './TabService.vue'
import TabProcess   from './TabProcess.vue'
import TabPortfolio from './TabPortfolio.vue'

// 탭 데이터 — label(화면 표시), id(식별자), component(연결할 컴포넌트)
// component에 import한 컴포넌트 객체를 직접 담음
const tabs = [
  { id: 'service',   label: '서비스 소개', component: TabService   },
  { id: 'process',   label: '진행 프로세스', component: TabProcess   },
  { id: 'portfolio', label: '포트폴리오', component: TabPortfolio },
]

// ref — 현재 선택된 탭 ID
// 초기값으로 첫 번째 탭 ID를 넣어둠
const currentTab = ref('service')

// computed — currentTab이 바뀔 때마다 자동 재계산
// tabs 배열에서 현재 탭 ID와 일치하는 항목을 찾아 컴포넌트 반환
// ?. (옵셔널 체이닝) — find 결과가 없을 경우 에러 방지
const currentComponent = computed(
  () => tabs.find(t => t.id === currentTab.value)?.component
)

📄 AccordionSection.vue
<ul class="accordion">

  <li
    v-for="(item, index) in list"
    :key="item.id"
    class="accordion__item"

    <!-- :class 배열 + 객체 혼합 방식
         'is-open' 클래스를 openIndex와 비교해서 조건부로 붙임 -->
    :class="{ 'is-open': openIndex === index }"
  >

    <!-- @click — 클릭한 아이템 index와 현재 열려있는 index 비교
         같은 거 클릭하면 닫고, 다른 거 클릭하면 그것만 열기 -->
    <button
      class="accordion__trigger"
      @click="toggle(index)"
    >
      <span>{{ item.question }}</span>
      <!-- 아이콘도 :class로 회전 처리 -->
      <svg
        class="accordion__icon"
        :class="{ 'is-rotated': openIndex === index }"
      >...</svg>
    </button>

    <!-- v-show — 열린 상태일 때만 표시
         아코디언 본문은 자주 토글되니까 v-if보다 v-show가 적합 -->
    <div
      class="accordion__body"
      v-show="openIndex === index"
    >
      <p>{{ item.answer }}</p>
    </div>

  </li>
</ul>
⚙️ AccordionSection.vue — script setup
import { ref } from 'vue'

const list = [
  { id: 1, question: '서비스 이용 방법은?', answer: '...' },
  { id: 2, question: '결제 수단은 어떤 게 있나요?', answer: '...' },
  { id: 3, question: '환불 정책이 궁금해요', answer: '...' },
]

// ref(null) — 현재 열려있는 항목의 index
// null이면 모두 닫힌 상태
const openIndex = ref(null)

// toggle 함수 — 같은 항목을 누르면 닫기(null), 다른 항목이면 그 index 저장
function toggle(index) {
  openIndex.value = openIndex.value === index ? null : index
}

Swiper 슬라이더 연동
template ref + onMounted + onUnmounted 조합
⚠️
Swiper 같은 외부 라이브러리는 실제 DOM 요소가 필요해요. Vue가 DOM을 만든 후인 onMounted 안에서만 초기화해야 해요. 그 전에 접근하면 null 에러 발생!
📄 MainBanner.vue — template
<section class="banner">

  <!-- ref="swiperEl" — 이 DOM 요소를 script의 swiperEl 변수에 연결
       onMounted 이후 swiperEl.value로 이 div에 접근 가능 -->
  <div ref="swiperEl" class="swiper">
    <div class="swiper-wrapper">

      <!-- v-for — slides 배열로 슬라이드 반복 생성
           실제 DOM이 만들어진 후 Swiper가 이 요소들을 인식함 -->
      <div
        v-for="slide in slides"
        :key="slide.id"
        class="swiper-slide"
      >
        <!-- :style — 슬라이드마다 다른 배경색 적용 -->
        <div class="banner__inner" :style="{ background: slide.bg }">
          <span class="banner__tag">{{ slide.tag }}</span>
          <h2 class="banner__title">{{ slide.title }}</h2>
          <p class="banner__desc">{{ slide.desc }}</p>

          <!-- :href — 동적 링크 바인딩 -->
          <a :href="slide.link" class="banner__cta">자세히 보기</a>
        </div>
      </div>

    </div>
    <div class="swiper-pagination"></div>
    <div class="swiper-button-prev"></div>
    <div class="swiper-button-next"></div>
  </div>

</section>

const sectionEl = ref(null)

// 배열 ref — v-for로 만들어진 여러 요소를 배열로 수집

const cards = [
  { id: 1, title: '서비스 A', desc: '설명', img: '/images/service-a.jpg' },
  { id: 2, title: '서비스 B', desc: '설명', img: '/images/service-b.jpg' },

import { ref, computed, onMounted } from 'vue'

const categories = [
  { label: '전체',   value: 'all'    },
  { label: '공지',   value: 'notice' },
  { label: '이벤트', value: 'event'  },
]

const list       = ref([])           // 전체 데이터
const currentCat = ref('all')        // 선택된 카테고리
const page       = ref(1)            // 현재 페이지 번호
const isLoading  = ref(false)        // 로딩 상태
const hasMore    = ref(true)         // 더 불러올 데이터가 있는지

// computed — currentCat이 바뀔 때마다 자동으로 필터링
// 'all'이면 전체 반환, 아니면 해당 카테고리만 필터링
const filteredList = computed(() => {
  if (currentCat.value === 'all') return list.value
  return list.value.filter(item => item.category === currentCat.value)
})

async function loadMore() {
  // 이미 로딩 중이면 중복 요청 방지
  if (isLoading.value) return
  isLoading.value = true

  try {
    // fetch API — 기본 JS, Vue 특유 문법 아님
       응답 받으면 list 배열에 추가 (스프레드로 기존 데이터 유지)
    const res  = await fetch(`/api/news?page=${page.value}`)
    const data = await res.json()

    list.value  = [...list.value, ...data.items]
    hasMore.value = data.hasNext
    page.value++
  } finally {
    // 성공/실패 상관없이 로딩 상태 해제
    isLoading.value = false
  }
}

// 컴포넌트 마운트 시 첫 페이지 로드
onMounted(() => loadMore())

import { ref, computed, reactive } from 'vue'

// reactive — 객체 전체를 반응형으로 만들 때 사용
// ref는 단일 값, reactive는 객체에 주로 씀
// form.name / form.email 처럼 . 으로 접근 (.value 불필요)
const form = reactive({
  name:     '',
  email:    '',
  category: '',
  message:  '',
  agreed:   false,
})

// errors — 각 필드 에러 메시지 저장
// 빈 문자열이면 에러 없음, 문자열이 있으면 에러 있음
const errors = reactive({
  name: '', email: '',
})

const isSubmitting = ref(false)

const categoryOptions = [
  { value: 'general', label: '일반 문의' },
  { value: 'service', label: '서비스 문의' },
  { value: 'partner', label: '제휴 문의' },
]

// computed — 폼 전체 유효성 검사
// 모든 조건이 충족될 때만 true → 버튼 활성화
// form 값이 바뀔 때마다 자동으로 재계산됨
const isFormValid = computed(() => {
  return form.name.length > 0
    && form.email.includes('@')
    && form.agreed
})

// 필드별 유효성 검사 — @blur 이벤트로 호출
// fieldName을 매개변수로 받아 공통 함수로 처리
function validate(fieldName) {
  if (fieldName === 'name') {
    errors.name = form.name ? '' : '이름을 입력해주세요'
  }
  if (fieldName === 'email') {
    errors.email = form.email.includes('@') ? '' : '올바른 이메일 형식이 아닙니다'
  }
}

async function handleSubmit() {
  isSubmitting.value = true
  try {
    await fetch('/api/contact', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      // reactive 객체는 JSON.stringify에 바로 사용 가능
      body: JSON.stringify(form),
    })
    alert('문의가 접수되었습니다.')
  } finally {
    isSubmitting.value = false
  }
}
